# Docker's docker-default AppArmor profile (moby/profiles, Apache-2.0) plus what Copilot CLI's sandbox (bubblewrap) needs:
# user namespaces and the mounts of its own root. It keeps docker-default's denies on /proc/sys, sysrq-trigger, kcore
# and /sys, so root in the container still cannot write them. Load it in Docker's virtual machine (the dev container
# guide, "Copilot CLI's sandbox") and name it with --security-opt apparmor=trackwall-sandbox.
#include <tunables/global>

profile trackwall-sandbox flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  network,
  capability,
  file,
  umount,
  mount,
  pivot_root,
  userns,
  signal (receive) peer=unconfined,
  signal (send,receive) peer=trackwall-sandbox,

  deny @{PROC}/* w,
  deny @{PROC}/{[^1-9],[^1-9][^0-9],[^1-9s][^0-9y][^0-9s],[^1-9][^0-9][^0-9][^0-9/]*}/** w,
  deny @{PROC}/sys/[^k]** w,
  deny @{PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{PROC}/sysrq-trigger rwklx,
  deny @{PROC}/kcore rwklx,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/fs/cg[^r]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,

  ptrace (trace,read,tracedby,readby) peer=trackwall-sandbox,
}
